RECOMMENDED IMPLEMENTATION
───────────────────────────────────────────────────────────────────────────

class WorkerWithHysteresis:
    def __init__(self):
        self.entry_threshold = 90    # Start tracking if Q >= 90
        self.exit_threshold = 85     # Stop tracking if Q < 85
        self.tracking = {}           # {symbol: 'ACTIVE' | 'INACTIVE'}

    def should_evaluate(self, symbol, quality):
        if symbol not in self.tracking:
            # NEW symbol - check entry threshold
            if quality >= self.entry_threshold:
                self.tracking[symbol] = 'ACTIVE'
                return True  # START tracking ✅
            return False

        elif self.tracking[symbol] == 'ACTIVE':
            # KNOWN symbol - check exit threshold
            if quality < self.exit_threshold:
                self.tracking[symbol] = 'INACTIVE'
                return False  # STOP tracking ❌
            return True  # CONTINUE tracking ✓

        return False
